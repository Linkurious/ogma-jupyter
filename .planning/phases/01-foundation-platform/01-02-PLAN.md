**js/src/types.ts:**
```typescript
// anywidget model interface
export interface Model {
    get(name: string): any;
    set(name: string, value: any): void;
    save_changes(): void;
    on(event: string, callback: () => void): void;
}

// Render context from anywidget
export interface RenderContext {
    model: Model;
    el: HTMLElement;
}

// Graph data structure matching Python traitlet
export interface GraphData {
    nodes: Array<{ id: string; data?: Record<string, any> }>;
    edges: Array<{ source: string; target: string; data?: Record<string, any> }>;
}
```

**js/src/index.ts:**
```typescript
import type { Model, RenderContext, GraphData } from './types';
// Note: Ogma import will fail until user configures npm registry
// During development, can use a mock or conditional import
import Ogma from '@linkurious/ogma';

function render({ model, el }: RenderContext) {
    // Set container size for visualization
    el.style.width = '100%';
    el.style.height = '400px';
    el.style.minHeight = '400px';

    // Initialize Ogma instance
    const ogma = new Ogma({
        container: el,
    });

    // Load initial graph data if present
    const loadGraph = () => {
        const data: GraphData = model.get('graph_data');
        if (data.nodes.length > 0) {
            ogma.setGraph({
                nodes: data.nodes.map(n => ({ id: n.id, data: n.data })),
                edges: data.edges.map(e => ({
                    source: e.source,
                    target: e.target,
                    data: e.data
                })),
            });
            // Center view on graph
            ogma.view.locateGraph();
        }
    };

    // Load initial data
    loadGraph();

    // React to graph_data changes from Python
    model.on('change:graph_data', loadGraph);

    // CRITICAL: Return cleanup function to prevent WebGL context leaks
    // Browsers limit WebGL contexts to 8-16; without cleanup, older
    // visualizations go blank when limit is reached
    return () => {
        ogma.destroy();
    };
}

export default { render };
```

**Important notes for implementation:**
- If Ogma npm registry isn't configured, the build will fail at import
- For initial development/testing, can create a mock Ogma or use try/catch
- The cleanup function is CRITICAL per RESEARCH.md pitfalls
- Container sizing ensures visualization is visible (not 0px height)
